#include <stdio.h>
 
long long calls = 0;
 
long long power_linear(long long base, int exp)
{
    calls++;
    if (exp == 0)
        return 1;
    return base * power_linear(base, exp - 1);
}
 
long long power_fast(long long base, int exp)
{
    long long half;
 
    calls++;
    if (exp == 0)
        return 1;
 
    half = power_fast(base, exp / 2);  /* compute only once */
    if (exp % 2 == 0)
        return half * half;
    return base * half * half;
}
 
int digital_root(int n)
{
    int sum = 0;
 
    if (n < 0)
        n = -n;
    if (n < 10)
        return n;
 
    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return digital_root(sum);
}
 
int main(void)
{
    long long a, b;
 
    calls = 0;
    a = power_linear(2, 30);
    printf("linear: %lld in %lld calls\n", a, calls);
 
    calls = 0;
    b = power_fast(2, 30);
    printf("fast: %lld in %lld calls\n", b, calls);
 
    printf("digital root of 9875: %d\n", digital_root(9875));
    printf("digital root of 0: %d\n", digital_root(0));
    printf("digital root of 1: %d\n", digital_root(1));
    return 0;
}
