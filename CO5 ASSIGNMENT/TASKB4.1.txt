#include <stdio.h>
#include <stdlib.h>
#include <string.h>
 
typedef struct {
    int id;
    char group[4];
} Donor;
 
typedef struct {
    Donor *data;
    int size;
    int capacity;
} DonorList;
 
int add_donor(DonorList *list, int id, const char *group)
{
    Donor *temp;
    int new_capacity;
 
    if (list == NULL || group == NULL)
        return 0;
 
    if (list->size == list->capacity) {
        new_capacity = (list->capacity == 0) ? 2 : list->capacity * 2;
 
        temp = realloc(list->data,
                       (size_t)new_capacity * sizeof(Donor));
        if (temp == NULL)
            return 0;  /* original list->data remains valid */
 
        list->data = temp;
        list->capacity = new_capacity;
    }
 
    list->data[list->size].id = id;
    strncpy(list->data[list->size].group, group, 3);
    list->data[list->size].group[3] = '\0';
    list->size++;
 
    return 1;
}
 
int main(void)
{
    DonorList list = {NULL, 0, 0};
    const char *groups[] = {"O+", "A+", "B-", "AB+", "O-"};
    int i;
 
    for (i = 0; i < 5; i++) {
        if (!add_donor(&list, 7001 + i, groups[i])) {
            fprintf(stderr, "Unable to allocate donor list\n");
            free(list.data);
            return 1;
        }
 
        printf("added %d (%s): size = %d, capacity = %d\n",
               list.data[i].id, list.data[i].group,
               list.size, list.capacity);
    }
 
    free(list.data);
    list.data = NULL;
    list.size = 0;
    list.capacity = 0;
    return 0;
}
