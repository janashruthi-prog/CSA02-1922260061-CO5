int *lowest_stock(int stock[], int n)
{
    int i, min_index = 0;
 
    if (stock == NULL || n <= 0)
        return NULL;
 
    for (i = 1; i < n; i++) {
        if (stock[i] < stock[min_index])
            min_index = i;
    }
    return &stock[min_index];
}
